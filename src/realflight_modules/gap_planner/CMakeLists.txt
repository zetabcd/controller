cmake_minimum_required(VERSION 3.16)
project(gap_planner)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(gap_msgs REQUIRED)
find_package(px4ctrl REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(yaml-cpp REQUIRED)
set(GCOPTER_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/../../../3rdpart/src/gcopter/gcopter/include"
  CACHE PATH "GCOPTER submodule include directory")
if(NOT EXISTS "${GCOPTER_INCLUDE}/gcopter/minco.hpp")
  message(FATAL_ERROR "Initialize 3rdpart/src/gcopter submodule before building gap_planner")
endif()
add_library(gap_planner_core src/geometry.cpp src/frame_geometry.cpp src/sparse_optimizer.cpp src/optimizer.cpp)
target_include_directories(gap_planner_core PUBLIC include)
target_include_directories(gap_planner_core SYSTEM PRIVATE "${GCOPTER_INCLUDE}" "${GCOPTER_INCLUDE}/gcopter")
target_link_libraries(gap_planner_core PUBLIC px4ctrl::px4ctrl_reference Eigen3::Eigen)
target_compile_options(gap_planner_core PRIVATE -O2 -Wall -Wextra)
add_executable(gap_planner_node src/planner_node.cpp)
target_link_libraries(gap_planner_node gap_planner_core)
ament_target_dependencies(gap_planner_node rclcpp gap_msgs geometry_msgs visualization_msgs std_msgs)
add_executable(gap_trajectory_sender src/sender_node.cpp)
target_link_libraries(gap_trajectory_sender gap_planner_core)
ament_target_dependencies(gap_trajectory_sender rclcpp gap_msgs)
add_executable(gap_plan_check src/plan_check.cpp)
target_link_libraries(gap_plan_check gap_planner_core yaml-cpp)
install(TARGETS gap_planner_node gap_trajectory_sender gap_plan_check DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/gap_console.py DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(gap_planner_flow_test test/planner_flow_test.cpp)
  target_link_libraries(gap_planner_flow_test gap_planner_core)
  ament_target_dependencies(gap_planner_flow_test rclcpp gap_msgs geometry_msgs visualization_msgs std_msgs)
  set_tests_properties(gap_planner_flow_test PROPERTIES ENVIRONMENT
    "ROS_DOMAIN_ID=189;ROS_LOCALHOST_ONLY=1;ROS_LOG_DIR=/tmp/gap_planner_flow_test")
  ament_add_gtest(gap_mocap_test test/mocap_input_test.cpp)
  target_include_directories(gap_mocap_test PRIVATE include)
  target_link_libraries(gap_mocap_test Eigen3::Eigen)
  ament_target_dependencies(gap_mocap_test geometry_msgs)
  ament_add_gtest(gap_geometry_test test/geometry_test.cpp)
  target_link_libraries(gap_geometry_test gap_planner_core)
  target_include_directories(gap_geometry_test SYSTEM PRIVATE "${GCOPTER_INCLUDE}")
endif()
ament_package()
